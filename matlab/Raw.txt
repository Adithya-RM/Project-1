clear
close all
clc

fs = 2000;
t = -1:1/fs:1;

rng(7)

normal = 0.7*randn(size(t));

normal = normal ...
    + 7*exp(-((t+0.68)/0.06).^2) ...
    - 4*exp(-((t+0.52)/0.03).^2) ...
    + 2*sin(2*pi*18*t).*exp(-((t+0.75)/0.2).^2);

athlete = 0.9*randn(size(t));

athlete = athlete ...
    + 2*sin(2*pi*12*t).*exp(-((t-0.15)/0.35).^2) ...
    + 1.5*sin(2*pi*25*t).*exp(-((t-0.30)/0.30).^2);

injured = 0.7*randn(size(t));

injured = injured ...
    + 6*exp(-((t+0.55)/0.08).^2) ...
    - 5*exp(-((t+0.42)/0.03).^2) ...
    + 4*exp(-((t-0.12)/0.05).^2) ...
    + 2.5*exp(-((t-0.28)/0.04).^2);

figure('Color','w','Position',[150 100 900 600])

subplot(3,1,1)
plot(t,normal,'b','LineWidth',1)
grid on
xlim([-1 1])
ylim([-6 11])
ylabel('Voltage (V)')
title('Raw — Normal')

subplot(3,1,2)
plot(t,athlete,'g','LineWidth',1)
grid on
xlim([-1 1])
ylim([-5 5])
ylabel('Voltage (V)')
title('Raw — Athlete')

subplot(3,1,3)
plot(t,injured,'r','LineWidth',1)
grid on
xlim([-1 0.8])
ylim([-10 10])
ylabel('Voltage (V)')
xlabel('Time (s)')
title('Raw — Injured Athlete')

sgtitle('VAG Raw Signals')

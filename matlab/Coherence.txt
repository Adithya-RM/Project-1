close all

fs = 2000;

window = hamming(256);
noverlap = 128;
nfft = 512;

[cxy1,f] = mscohere(normal,athlete,window,noverlap,nfft,fs);
[cxy2,~] = mscohere(normal,injured,window,noverlap,nfft,fs);
[cxy3,~] = mscohere(athlete,injured,window,noverlap,nfft,fs);

figure('Color','w','Position',[150 100 900 650])

subplot(3,1,1)
plot(f,cxy1,'b','LineWidth',1.2)
grid on
xlim([0 500])
ylim([0 1])
xlabel('Frequency (Hz)')
ylabel('Coherence')
title('Normal vs Athlete')

subplot(3,1,2)
plot(f,cxy2,'r','LineWidth',1.2)
grid on
xlim([0 500])
ylim([0 1])
xlabel('Frequency (Hz)')
ylabel('Coherence')
title('Normal vs Injured')

subplot(3,1,3)
plot(f,cxy3,'k','LineWidth',1.2)
grid on
xlim([0 500])
ylim([0 1])
xlabel('Frequency (Hz)')
ylabel('Coherence')
title('Athlete vs Injured')

sgtitle('Magnitude-Squared Coherence')

fprintf('\nMean Coherence Values\n')
fprintf('Normal vs Athlete : %.4f\n',mean(cxy1))
fprintf('Normal vs Injured : %.4f\n',mean(cxy2))
fprintf('Athlete vs Injured: %.4f\n',mean(cxy3))

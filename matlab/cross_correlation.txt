[c2,~] = xcorr(normal,injured,'coeff');
[c3,~] = xcorr(athlete,injured,'coeff');

figure('Color','w','Position',[150 100 900 650])

plot(lags,c1,'b','LineWidth',1.2)
grid on
xlim([-1 1])
ylim([-0.3 1])
xlabel('Lag (s)')
ylabel('Correlation')
title('Normal vs Athlete')

xlabel('Lag (s)')
ylabel('Correlation')
title('Normal vs Injured')

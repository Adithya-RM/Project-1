close all

fs = 2000;

window = hamming(256);
noverlap = 128;
nfft = 512;

[Pn,f] = pwelch(normal,window,noverlap,nfft,fs);
[Pa,~] = pwelch(athlete,window,noverlap,nfft,fs);
[Pi,~] = pwelch(injured,window,noverlap,nfft,fs);

figure('Color','w','Position',[150 100 850 500])

plot(f,10*log10(Pn),'LineWidth',1.3)
hold on
plot(f,10*log10(Pa),'LineWidth',1.3)
plot(f,10*log10(Pi),'LineWidth',1.3)

grid on
xlim([0 500])
ylim([-65 -30])

xlabel('Frequency (Hz)')
ylabel('Power (dB)')
title('PSD Comparison — All Three Signals')

legend('Normal','Athlete','Injured Athlete','Location','northeast')
